## Global Constraints

- 套餐定价货币为 CNY（人民币），单位为「分」存储、「元」展示：尝鲜价 990 分 = ¥9.9。
- 套餐 ID 全部小写下划线：`earlybird_9_9`、`pro_monthly`、`pro_yearly`。
- 尝鲜价套餐必须带有效期窗口（`start_at..end_at`，aware UTC，含秒级）与限量（`max_redemptions`），售罄或过期后不可再开通。
- 不引入任何新外部依赖；仅用标准库，保持 `pyproject.toml` 不变。
- 时间一律以 `datetime`（aware，`zoneinfo.utc`）参与比较与持久化。
- 金额抵扣/退还一律整数分运算，禁止 `float`；升级抵扣按「剩余天数比例」线性计算。
- 现有 `market/` 模块不被修改（零侵入）；套餐模块对 `market/` 仅单向依赖（如需查询赠送技能）。
- 遵循现有 `market/schema.py` 的 `@dataclass(frozen=True)` 风格与 `# -*- coding: utf-8 -*-` 文件头。
- 提交粒度遵循 DRY/YAGNI/TDD：每个 Task 以 failing test 起步、最小实现通过、单独 commit。

```python
# src/qwenpaw/pricing/service.py
# -*- coding: utf-8 -*-
"""Pricing service: validate, redeem, quota, upgrade."""
from __future__ import annotations
from datetime import datetime, timedelta, timezone
from typing import Callable

from .errors import PlanUnavailable, RedemptionExhausted, WindowClosed
from .schema import EarlyBirdTier, PlanId, Redemption
from .tiers import CATALOG, earlybird_for


class PricingService:
    def __init__(self, now: Callable[[], datetime] | None = None) -> None:
        self._now = now or (lambda: datetime.now(timezone.utc))
        self._redeemed: dict[str, Redemption] = {}
        self._counts: dict[PlanId, int] = {}

    def is_window_open(self, eb: EarlyBirdTier) -> bool:
        now = self._now()
        return eb.start_at <= now <= eb.end_at

    def is_exhausted(self, eb: EarlyBirdTier) -> bool:
        return self._counts.get(eb.tier.plan_id, 0) >= eb.max_redemptions

    def redeem(self, plan_id: PlanId, user_id: str) -> Redemption:
        tier = CATALOG.get(plan_id)
        if tier is None:
            raise PlanUnavailable(f"unknown plan: {plan_id}")
        eb = earlybird_for(plan_id)
        if eb is not None:
            if not self.is_window_open(eb):
                raise WindowClosed("early-bird window not open")
            if self.is_exhausted(eb):
                raise RedemptionExhausted("early-bird redemptions exhausted")
        now = self._now()
        redemption = Redemption(
            plan_id=plan_id,
            user_id=user_id,
            redeemed_at=now,
            expires_at=now + timedelta(days=tier.period_days),
        )
        self._redeemed[user_id] = redemption
        self._counts[plan_id] = self._counts.get(plan_id, 0) + 1
        return redemption
```

def upgrade_estimate(self, from_id: PlanId, to_id: PlanId, redeemed_at: datetime) -> UpgradeEstimate:
    from_tier = CATALOG.get(from_id)
    to_tier = CATALOG.get(to_id)
    if from_tier is None or to_tier is None:
        raise PlanUnavailable("unknown plan in upgrade")
    now = self._now()
    total = from_tier.period_days
    remaining = max(0, (redeemed_at + timedelta(days=total) - now).days)
    credit = from_tier.price_cents * remaining // total if total else 0
    pay = max(0, to_tier.price_cents - credit)
    return UpgradeEstimate(from_plan=from_id, to_plan=to_id, credit_cents=credit, pay_cents=pay)
```

# src/qwenpaw/cli/pricing_cmd.py
# -*- coding: utf-8 -*-
"""CLI for pricing & plans."""
from __future__ import annotations
import argparse

from ..pricing.schema import PlanId
from ..pricing.service import PricingService
from ..pricing.tiers import CATALOG, EARLYBIRD_9_9


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="dtcoder pricing", description="套餐与定价")
    sub = p.add_subparsers(dest="command", required=True)

    sub.add_parser("plans", help="列出所有套餐")
    r = sub.add_parser("redeem", help="兑换套餐")
    r.add_argument("plan_id", choices=[pid.value for pid in PlanId])
    r.add_argument("--user", required=True)

    u = sub.add_parser("upgrade", help="升级抵扣估算")
    u.add_argument("--from", dest="from_plan", required=True)
    u.add_argument("--to", dest="to_plan", required=True)

    sub.add_parser("status", help="查看尝鲜价剩余")
    return p


def main(argv: list[str] | None = None) -> int:
    ns = build_parser().parse_args(argv)
    svc = PricingService()
    if ns.command == "plans":
        for pid, tier in CATALOG.items():
            print(f"{pid.value}\t{tier.price_cents/100:.2f}\t{tier.label}")
    elif ns.command == "redeem":

        r = svc.redeem(plan, ns.user)
        print(f"redeemed {r.plan_id.value} for {r.user_id}, expires {r.expires_at.isoformat()}")
    elif ns.command == "status":
        print(f"earlybird 9.9: {EARLYBIRD_9_9.redeemed_count}/{EARLYBIRD_9_9.max_redemptions}")
    return 0
```
